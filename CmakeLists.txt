cmake_minimum_required(VERSION 3.15)
project(MiserableTree VERSION 1.0 LANGUAGES CXX)

# Enforce C++17 standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Set compiler warning levels based on platform
if(MSVC)
    add_compile_options(/W4)
else()
    add_compile_options(-Wall -Wextra)
endif()

# Recursively find all source and correctness test files
file(GLOB_RECURSE SRC_FILES "src/*.cpp")
file(GLOB_RECURSE TEST_FILES "tests/*.cpp")

# Define main application executable
add_executable(program main.cpp ${SRC_FILES})

# Define correctness test runner (Excludes benchmark files automatically)
add_executable(run_tests ${SRC_FILES} ${TEST_FILES})
target_include_directories(run_tests PRIVATE src)

# Define scale performance benchmark executable
add_executable(scale_benchmark "benchmark/benchmark.cpp" ${SRC_FILES})
target_include_directories(scale_benchmark PRIVATE src)

# Link Windows Process Status API (psapi) for RAM profiling
if(WIN32)
    target_link_libraries(scale_benchmark PRIVATE psapi)
endif()

# Copy dataset folder to the build directory automatically
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/data" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")

# Custom target to completely wipe out the build directory
add_custom_target(clean-all
    COMMAND ${CMAKE_COMMAND} -E remove_directory "${CMAKE_BINARY_DIR}"
    COMMENT "Cleaning entire build directory..."
)
